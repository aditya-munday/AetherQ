find_package(Threads REQUIRED)

function(aetherq_add_test name category)
  add_executable(${name} ${ARGN})
  target_link_libraries(${name} PRIVATE aetherq GTest::gtest GTest::gtest_main aetherq_warnings)
  target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  add_test(NAME ${name} COMMAND ${name})
  set_tests_properties(${name} PROPERTIES LABELS "${category}")
endfunction()

aetherq_add_test(unit_memory unit unit/test_memory.cpp)
aetherq_add_test(unit_executor unit unit/test_executor.cpp)
aetherq_add_test(unit_config unit unit/test_config.cpp)

aetherq_add_test(numerical_workloads numerical numerical/test_numerical_workloads.cpp)
